cmake_minimum_required(VERSION 3.21)
project(xing_web LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(USE_SANDBOX OFF CACHE BOOL "CEF distribution without the sandbox library")
set(PROJECT_ARCH x86_64)
if(NOT CEF_ROOT)
    file(GLOB CEF_SDKS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/cef_binary_*_windows64_minimal")
    list(LENGTH CEF_SDKS SDK_COUNT)
    if(NOT SDK_COUNT EQUAL 1)
        message(FATAL_ERROR "Set CEF_ROOT to the Windows x64 minimal SDK")
    endif()
    list(GET CEF_SDKS 0 CEF_ROOT)
endif()
list(APPEND CMAKE_MODULE_PATH "${CEF_ROOT}/cmake")
find_package(CEF REQUIRED)
find_package(JNI REQUIRED)
add_subdirectory("${CEF_LIBCEF_DLL_WRAPPER_PATH}" cef_wrapper)
ADD_LOGICAL_TARGET("libcef_lib" "${CEF_LIB_DEBUG}" "${CEF_LIB_RELEASE}")
add_library(xing-web SHARED WebEngine.cpp Projection.cpp)
add_executable(xing-web-process WIN32 Subprocess.cpp)
foreach(target xing-web xing-web-process)
    SET_EXECUTABLE_TARGET_PROPERTIES(${target})
    target_include_directories(${target} PRIVATE "${CEF_ROOT}" ${JNI_INCLUDE_DIRS})
    target_compile_definitions(${target} PRIVATE _DISABLE_CONSTEXPR_MUTEX_CONSTRUCTOR)
    target_link_libraries(${target} PRIVATE libcef_lib libcef_dll_wrapper ${CEF_STANDARD_LIBS})
    set_target_properties(${target} PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
endforeach()
target_link_libraries(xing-web PRIVATE delayimp)
target_link_options(xing-web PRIVATE /DELAYLOAD:libcef.dll)
file(MAKE_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
file(COPY "${CEF_ROOT}/Release/" "${CEF_ROOT}/Resources/" DESTINATION "${CMAKE_CURRENT_SOURCE_DIR}/runtime")
configure_file("${CEF_ROOT}/LICENSE.txt" "${CMAKE_CURRENT_SOURCE_DIR}/runtime/CEF-LICENSE.txt" COPYONLY)
